# 在阿里云 Workbench 中复制下面整段到终端执行
# 用于诊断档案照片保存失败

cd ~/gemini-audio-service

echo "========== 档案照片保存诊断 =========="
echo ""

echo "=== 1. 最近档案/照片相关日志 ==="
tail -200 ~/gemini-audio-service.log 2>/dev/null | grep -E "\[档案照片\]|\[档案更新\]|upload-photo|photo_url|OSS上传" | tail -25

echo ""
echo "=== 2. 最近 ERROR/WARNING ==="
tail -100 ~/gemini-audio-service.log 2>/dev/null | grep -E "ERROR|WARNING|Exception|失败" | tail -15

echo ""
echo "=== 3. OSS 配置 ==="
grep -E "USE_OSS|OSS_BUCKET|OSS_ENDPOINT" .env 2>/dev/null | sed 's/\(=.\)[^=]*/\1***/' || echo "未找到 .env 或 OSS 配置"

echo ""
echo "=== 4. 健康检查 ==="
curl -sf --max-time 5 http://127.0.0.1:8000/health && echo " OK" || echo " FAIL"

echo ""
echo "========== 诊断完成 =========="
echo ""
echo "若 [档案照片] OSS上传返回None → 检查 OSS 配置、bucket 权限"
echo "若 [档案更新] 档案不存在 → 检查 profile_id 是否属于当前用户"
echo "若 HTTP 401/500 → 查看上方日志详情"
