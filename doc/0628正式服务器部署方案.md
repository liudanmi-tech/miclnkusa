## 二、一键部署脚本

将以下内容保存为 `deploy_test_to_prod.sh`，放在本地任意目录，执行 `bash deploy_test_to_prod.sh` 即可。

```bash
#!/bin/bash
set -e   # 任意步骤失败立即停止

TEST_HOST="34.74.255.48"
PROD_HOST="34.74.150.225"
USER="liudanmi"
SERVICE_DIR="/opt/gemini-audio-service"
TEMP_DIR="/tmp/gemini_deploy_$(date +%Y%m%d_%H%M%S)"

echo ""
echo "=================================================="
echo "  测试服 → 正式服 一键部署"
echo "  $(date '+%Y-%m-%d %H:%M:%S')"
echo "=================================================="

# ──────────────────────────────────────────────────
# Step 1: 从测试服拉取代码到本地临时目录
# ──────────────────────────────────────────────────
echo ""
echo "【Step 1】从测试服拉取代码..."
mkdir -p "$TEMP_DIR"

rsync -avz --progress \
  --exclude='.env' \
  --exclude='.venv' \
  --exclude='__pycache__' \
  --exclude='*.pyc' \
  --exclude='*.pyc' \
  --exclude='gemini-audio-service.log' \
  -e ssh \
  "$USER@$TEST_HOST:$SERVICE_DIR/" \
  "$TEMP_DIR/"

echo "✅ 代码拉取完成，保存到 $TEMP_DIR"

# ──────────────────────────────────────────────────
# Step 2: 将代码推到正式服临时目录
# ──────────────────────────────────────────────────
echo ""
echo "【Step 2】将代码推送到正式服..."

rsync -avz --progress \
  --exclude='.env' \
  --exclude='.venv' \
  --exclude='__pycache__' \
  --exclude='*.pyc' \
  -e ssh \
  "$TEMP_DIR/" \
  "$USER@$PROD_HOST:~/deploy_temp/"

echo "✅ 代码已传到正式服 ~/deploy_temp/"

# ──────────────────────────────────────────────────
# Step 3: 在正式服 sudo 覆盖服务目录（保护 .env）
# ──────────────────────────────────────────────────
echo ""
echo "【Step 3】在正式服覆盖服务代码..."

ssh "$USER@$PROD_HOST" "
  echo '备份正式服 .env...'
  sudo cp $SERVICE_DIR/.env /tmp/.env_prod_backup_\$(date +%Y%m%d_%H%M%S) 2>/dev/null || true

  echo '覆盖代码文件...'
  sudo rsync -av \
    --exclude='.env' \
    --exclude='.venv' \
    --exclude='__pycache__' \
    ~/deploy_temp/ \
    $SERVICE_DIR/

  echo '还原 .env...'
  sudo cp /tmp/.env_prod_backup_* $SERVICE_DIR/.env 2>/dev/null | head -1 || true
"

echo "✅ 代码覆盖完成，.env 已保护"

# ──────────────────────────────────────────────────
# Step 4: 正式服数据库迁移
# ──────────────────────────────────────────────────
echo ""
echo "【Step 4】执行数据库迁移..."

ssh "$USER@$PROD_HOST" "
  # 4a: 给 sessions 表补全字段
  echo '补充 sessions 表字段...'
  sudo -u postgres psql gemini_audio_db << 'EOSQL'
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS session_type VARCHAR(20) DEFAULT 'review';
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS card_title VARCHAR(100);
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS live_summary TEXT;
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS summary_status VARCHAR(20) DEFAULT 'pending';
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS skills_status VARCHAR(20) DEFAULT 'pending';
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS cover_image_url TEXT;
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS total_images_expected INTEGER DEFAULT 0;
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS total_images_completed INTEGER DEFAULT 0;
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS total_images_failed INTEGER DEFAULT 0;
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS gemini_token TEXT;
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS gemini_token_updated TIMESTAMPTZ;
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS ws_disconnected_at TIMESTAMPTZ;
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS ended_at TIMESTAMPTZ;
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS abandon_reason VARCHAR(100);
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS finalize_status VARCHAR(20) DEFAULT 'pending';
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS mood_state VARCHAR(50);
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS emotion_type VARCHAR(100);
ALTER TABLE sessions ADD COLUMN IF NOT EXISTS emotion_intensity INTEGER;
EOSQL

  # 4b: 给 profiles 表补全声纹字段
  echo '补充 profiles 表字段...'
  sudo -u postgres psql gemini_audio_db << 'EOSQL'
ALTER TABLE profiles ADD COLUMN IF NOT EXISTS voiceprint_intent BOOLEAN DEFAULT FALSE;
ALTER TABLE profiles ADD COLUMN IF NOT EXISTS voiceprint_intent_at TIMESTAMPTZ;
ALTER TABLE profiles ADD COLUMN IF NOT EXISTS voice_embedding FLOAT[];
ALTER TABLE profiles ADD COLUMN IF NOT EXISTS voice_embedding_updated_at TIMESTAMPTZ;
ALTER TABLE profiles ADD COLUMN IF NOT EXISTS voice_embedding_zh FLOAT[];
ALTER TABLE profiles ADD COLUMN IF NOT EXISTS voice_embedding_zh_updated_at TIMESTAMPTZ;
EOSQL

  # 4c: 给 strategy_analysis 补全字段
  echo '补充 strategy_analysis 表字段...'
  sudo -u postgres psql gemini_audio_db -c \"ALTER TABLE strategy_analysis ADD COLUMN IF NOT EXISTS conversation JSONB;\"

  # 4d: 跑 Live Mode 迁移脚本（建 7 张新表）
  echo '运行 Live Mode 迁移脚本...'
  cd $SERVICE_DIR
  sudo .venv/bin/python database/migrations/run_add_live_mode_tables.py
"

echo "✅ 数据库迁移完成"

# ──────────────────────────────────────────────────
# Step 5: 重启服务
# ──────────────────────────────────────────────────
echo ""
echo "【Step 5】重启正式服服务..."

ssh "$USER@$PROD_HOST" "
  sudo systemctl restart gemini-audio.service
  sleep 3
  sudo systemctl status gemini-audio.service --no-pager | head -20
"

echo "✅ 服务已重启"

# ──────────────────────────────────────────────────
# Step 6: 验证部署结果
# ──────────────────────────────────────────────────
echo ""
echo "【Step 6】验证部署结果..."

echo -n "  正式服健康检查: "
HTTP_STATUS=$(curl -s -o /dev/null -w "%{http_code}" --max-time 10 "http://34.74.150.225/api/v1/app-config" || echo "000")
if [ "$HTTP_STATUS" = "200" ] || [ "$HTTP_STATUS" = "401" ]; then
  echo "✅ 服务正常响应 (HTTP $HTTP_STATUS)"
else
  echo "❌ 服务响应异常 (HTTP $HTTP_STATUS)，请检查日志"
fi

# 清理本地临时文件
rm -rf "$TEMP_DIR"
echo ""
echo "=================================================="
echo "  部署完成！$(date '+%Y-%m-%d %H:%M:%S')"
echo "=================================================="
echo ""
echo "后续验证建议："
echo "  1. 打开 App 切换到正式服，测试登录、录音、AI 聊天"
echo "  2. 查看正式服日志：ssh liudanmi@34.74.150.225 'sudo journalctl -u gemini-audio.service -f'"
echo ""
```

## 四、回滚方案

如果部署后正式服出现严重问题：

```bash
# 1. 回滚代码：正式服上恢复 .env 备份并重启
ssh liudanmi@34.74.150.225 "
  sudo cp /tmp/.env_prod_backup_* /opt/gemini-audio-service/.env
  sudo systemctl restart gemini-audio.service
"

# 2. 数据库字段是用 ADD COLUMN IF NOT EXISTS 加的，不影响原有数据，无需回滚
# 3. 新增的 Live Mode 表为空表，如需删除：
ssh liudanmi@34.74.150.225 "sudo -u postgres psql gemini_audio_db -c '
  DROP TABLE IF EXISTS live_turns CASCADE;
  DROP TABLE IF EXISTS live_events CASCADE;
  DROP TABLE IF EXISTS live_segments CASCADE;
  DROP TABLE IF EXISTS live_speaker_mappings CASCADE;
  DROP TABLE IF EXISTS live_summary_tasks CASCADE;
  DROP TABLE IF EXISTS live_panel_batches CASCADE;
'"
```

## 六、常见问题

**Q: 部署后用户订阅状态消失？**
A: 正式服和测试服是独立数据库，用户需重新登录。已有正式服账号数据不受影响。

**Q: Live Mode 脚本报 already exists？**
A: 正常，脚本有容错处理，会跳过已存在的表并继续验证。

**Q: rsync 报 Permission denied？**
A: 确认本地 `~/.ssh/id_rsa` 对两台服务器都有效，运行前置条件检查命令。
